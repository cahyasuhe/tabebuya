Setup Google Apps Script (Multi-Role & MD5 Password)

Salin kode Google Apps Script di bawah ini ke Spreadsheet RT Anda. Script ini mendukung sheet Warga, Pegawai, Iuran, dan Kas. 

function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var dataWarga = getSheetDataAsObjects(ss.getSheetByName("Warga"));
  var dataPegawai = getSheetDataAsObjects(ss.getSheetByName("Pegawai"));
  var dataIuran = getSheetDataAsObjects(ss.getSheetByName("Iuran"));
  var dataKas = getSheetDataAsObjects(ss.getSheetByName("Kas"));
  
  return ContentService.createTextOutput(JSON.stringify({
    status: "success",
    warga: dataWarga,
    pegawai: dataPegawai,
    iuran: dataIuran,
    kas: dataKas
  })).setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    var contents = JSON.parse(e.postData.contents);
    var action = contents.action;
    var ss = SpreadsheetApp.getActiveSpreadsheet();

    if (action === "ADD_WARGA") {
      var w = contents.data;
      var sheetWarga = ss.getSheetByName("Warga");
      sheetWarga.appendRow([w.id, w.gang, w.noRumah, w.nama, w.hp, w.statusHuni, w.username || w.id, w.password || ""]);
      return responseJSON({ status: "success", message: "Warga disimpan" });
    }
    if (action === "RECORD_IURAN") {
      var i = contents.data;
      ss.getSheetByName("Iuran").appendRow([i.key, i.wargaId, i.bulan, i.tahun, i.nominal, i.date]);
      return responseJSON({ status: "success", message: "Iuran disimpan" });
    }
    if (action === "ADD_KAS") {
      var k = contents.data;
      ss.getSheetByName("Kas").appendRow([k.id, k.date, k.type, k.category, k.note, k.amount, k.officer]);
      return responseJSON({ status: "success", message: "Kas disimpan" });
    }
    return responseJSON({ status: "error", message: "Aksi tidak dikenal" });
  } catch (err) {
    return responseJSON({ status: "error", message: err.toString() });
  }
}

function getSheetDataAsObjects(sheet) {
  if (!sheet) return [];
  var data = sheet.getDataRange().getValues();
  if (data.length <= 1) return [];
  var headers = data[0];
  var rows = [];
  for (var i = 1; i < data.length; i++) {
    var obj = {};
    for (var j = 0; j < headers.length; j++) obj[headers[j]] = data[i][j];
    rows.push(obj);
  }
  return rows;
}

function responseJSON(data) {
  return ContentService.createTextOutput(JSON.stringify(data)).setMimeType(ContentService.MimeType.JSON);
}

Panduan Complete Deployment & Hosting Aplikasi Kas RT

Langkah publikasi aplikasi ke internet tanpa biaya server (100% Gratis).

Metode 1: GitHub Pages
  1. Buat Repository baru di GitHub.com.
  2. Upload file ini dan beri nama index.html.
  3. Buka menu Settings → Pages.
  4. Pada bagian Branch, pilih main dan simpan (Save).
  5. Aplikasi langsung dapat diakses secara publik via URL github.io!
Metode 2: Netlify / Vercel (Drag & Drop)
  1. Buka app.netlify.com/drop atau vercel.com.
  2. Tarik (Drag & Drop) folder berisi file index.html ini.
  3. Tunggu proses publish dalam 5 detik.
  4. Domain gratis dengan HTTPS otomatis didapatkan.
