Setup Google Apps Script (Multi-Role & Auto-Sheet)

Salin kode Google Apps Script terbaru di bawah ini. Kode ini telah disesuaikan untuk mendukung Tabebuya Raya, Tabebuya 1, Tabebuya 2, dan Tabebuya 3, serta pemisahan transfer langsung ke Bendahara Umum. 

function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  
  var sheetWarga = getOrCreateSheet(ss, "Warga", ["id", "gang", "noRumah", "nama", "hp", "statusHuni", "username", "password"]);
  var sheetPegawai = getOrCreateSheet(ss, "Pegawai", ["id", "username", "password", "nama", "role", "linkedGang"]);
  var sheetIuran = getOrCreateSheet(ss, "Iuran", ["key", "wargaId", "bulan", "tahun", "nominal", "date", "handedOver", "method", "collector"]);
  var sheetKas = getOrCreateSheet(ss, "Kas", ["id", "date", "type", "category", "note", "amount", "officer"]);

  initDefaultPegawai(sheetPegawai);

  return responseJSON({
    status: "success",
    warga: getSheetDataAsObjects(sheetWarga),
    pegawai: getSheetDataAsObjects(sheetPegawai),
    iuran: getSheetDataAsObjects(sheetIuran),
    kas: getSheetDataAsObjects(sheetKas)
  });
}

function doPost(e) {
  try {
    var contents = JSON.parse(e.postData.contents);
    var action = contents.action;
    var data = contents.data;
    var ss = SpreadsheetApp.getActiveSpreadsheet();

    if (action === "ADD_WARGA") {
      var sheet = getOrCreateSheet(ss, "Warga", ["id", "gang", "noRumah", "nama", "hp", "statusHuni", "username", "password"]);
      sheet.appendRow([data.id, data.gang, data.noRumah, data.nama, data.hp, data.statusHuni, data.username || data.nama, data.password || "123456"]);
      return responseJSON({ status: "success", message: "Warga berhasil ditambahkan" });
    }

    if (action === "UPDATE_WARGA") {
      var sheet = ss.getSheetByName("Warga");
      if (sheet) {
        var rows = sheet.getDataRange().getValues();
        for (var i = 1; i < rows.length; i++) {
          if (String(rows[i][0]) === String(data.id)) {
            sheet.getRange(i + 1, 1, 1, 8).setValues([[
              data.id, data.gang, data.noRumah, data.nama, data.hp, data.statusHuni, data.username || data.nama, data.password || rows[i][7] || "123456"
            ]]);
            return responseJSON({ status: "success", message: "Data warga diperbarui" });
          }
        }
      }
      return responseJSON({ status: "error", message: "Warga tidak ditemukan" });
    }

    if (action === "DELETE_WARGA") {
      var sheet = ss.getSheetByName("Warga");
      if (sheet) {
        var rows = sheet.getDataRange().getValues();
        for (var i = 1; i < rows.length; i++) {
          if (String(rows[i][0]) === String(data.id)) {
            sheet.deleteRow(i + 1);
            return responseJSON({ status: "success", message: "Warga berhasil dihapus" });
          }
        }
      }
      return responseJSON({ status: "error", message: "Warga tidak ditemukan" });
    }

    if (action === "RECORD_IURAN") {
      var sheet = getOrCreateSheet(ss, "Iuran", ["key", "wargaId", "bulan", "tahun", "nominal", "date", "handedOver", "method", "collector"]);
      // Jika pembayaran TRANSFER langsung ke Bendahara Umum, handedOver otomatis true
      var isDirectTransfer = (data.method === "TRANSFER");
      var handedOverValue = isDirectTransfer ? true : (data.handedOver || false);

      sheet.appendRow([
        data.key, 
        data.wargaId, 
        data.bulan, 
        data.tahun, 
        data.nominal, 
        data.date, 
        handedOverValue, 
        data.method || "TUNAI", 
        data.collector || "Bendahara Gang"
      ]);

      return responseJSON({ status: "success", message: "Iuran berhasil dicatat" });
    }

    if (action === "HANDOVER_GANG") {
      var sheet = ss.getSheetByName("Iuran");
      if (sheet) {
        var rows = sheet.getDataRange().getValues();
        for (var i = 1; i < rows.length; i++) {
          if (data.keys && data.keys.indexOf(String(rows[i][0])) !== -1) {
            sheet.getRange(i + 1, 7).setValue(true); // Set handedOver = true
          }
        }
      }
      return responseJSON({ status: "success", message: "Saldo kas gang berhasil diserahkan ke Bendahara Umum" });
    }

    if (action === "ADD_KAS") {
      var sheet = getOrCreateSheet(ss, "Kas", ["id", "date", "type", "category", "note", "amount", "officer"]);
      sheet.appendRow([data.id || Date.now(), data.date, data.type, data.category, data.note, data.amount, data.officer]);
      return responseJSON({ status: "success", message: "Transaksi kas disimpan" });
    }

    return responseJSON({ status: "error", message: "Aksi tidak dikenal" });
  } catch (err) {
    return responseJSON({ status: "error", message: err.toString() });
  }
}

function getOrCreateSheet(ss, name, headers) {
  var sheet = ss.getSheetByName(name);
  if (!sheet) {
    sheet = ss.insertSheet(name);
    sheet.appendRow(headers);
    sheet.getRange(1, 1, 1, headers.length).setFontWeight("bold").setBackground("#e0f2fe");
  }
  return sheet;
}

function initDefaultPegawai(sheet) {
  if (sheet.getLastRow() <= 1) {
    sheet.appendRow(["PEG-01", "bendahara_raya", "123456", "Bendahara Tabebuya Raya", "bendahara_gang", "Tabebuya Raya"]);
    sheet.appendRow(["PEG-02", "bendahara1", "123456", "Bendahara Tabebuya 1", "bendahara_gang", "Tabebuya 1"]);
    sheet.appendRow(["PEG-03", "bendahara2", "123456", "Bendahara Tabebuya 2", "bendahara_gang", "Tabebuya 2"]);
    sheet.appendRow(["PEG-04", "bendahara3", "123456", "Bendahara Tabebuya 3", "bendahara_gang", "Tabebuya 3"]);
    sheet.appendRow(["PEG-05", "bendahara_umum", "123456", "Bendahara Umum RT", "bendahara_umum", ""]);
  }
}

function getSheetDataAsObjects(sheet) {
  if (!sheet) return [];
  var data = sheet.getDataRange().getValues();
  if (data.length <= 1) return [];
  var headers = data[0];
  var rows = [];
  for (var i = 1; i < data.length; i++) {
    var obj = {};
    for (var j = 0; j < headers.length; j++) obj[headers[j]] = data[i][j];
    rows.push(obj);
  }
  return rows;
}

function responseJSON(data) {
  return ContentService.createTextOutput(JSON.stringify(data)).setMimeType(ContentService.MimeType.JSON);
}
